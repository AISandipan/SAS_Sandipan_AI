/* Advanced Neural Network Models in SAS */

/*----------------------------------------------------*/
/* 1. Customer Churn Prediction Model */
/*----------------------------------------------------*/
proc neural data=work.Customer_Churn;
   input Age Balance Tenure Products CreditScore / level=interval;
   target Churn / level=nominal;

   hidden 5;
   hidden 3;

   train maxiter=100 outmodel=Churn_Model;
run;


/*----------------------------------------------------*/
/* 2. Fraud Detection Neural Network */
/*----------------------------------------------------*/
proc neural data=work.Fraud_Data;
   input Amount Frequency Transactions AvgSpend / level=interval;
   target FraudFlag / level=nominal;

   hidden 6;

   train maxiter=150 outmodel=Fraud_Model;
run;

/* 17. Rank Accounts by Total Visits with error code..
SAS does not allow calculated columns (like calculated Total_Visit)
to be referenced inside a window function (like RANK()).
SAS SQL does not support subqueries in the FROM clause (inline views)
when using window functions directly like in standard SQL. 
It has limitations compared to full ANSI SQL
Also, SAS only supports window functions (like RANK() OVER(...)) in certain contexts

*/
proc sql;
    create table Ranked_Accounts as
    select ACCOUNT, sum(VISIT) as Total_Visit,
           rank() over (order by calculated Total_Visit desc) as Visit_Rank
    from Bank
    group by ACCOUNT;
quit;


proc sql;
    create table Ranked_Accounts as
    select ACCOUNT, Total_Visit,
           rank() over (order by Total_Visit desc) as Visit_Rank
    from (
        select ACCOUNT, sum(VISIT) as Total_Visit
        from Bank
        group by ACCOUNT
    );
quit;
/* Solution for above error.... */
proc sql;
    create table Total_Visits as
    select ACCOUNT, sum(VISIT) as Total_Visit
    from Bank
    group by ACCOUNT;
quit;

proc rank data=Total_Visits out=Ranked_Accounts descending;
    var Total_Visit;
    ranks Visit_Rank;
run;

proc sort data=Ranked_Accounts out=Sorted_Accounts;

run;

proc sort data=Ranked_Accounts out=Sorted_Accounts;

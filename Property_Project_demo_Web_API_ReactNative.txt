// Infrastructure / Repositories / ICartRepository.cs
public interface ICartRepository
{
    Task<Cart> GetCartAsync(Guid cartId);
    Task<Cart> CreateCartAsync();
    Task AddOrUpdateItemAsync(Guid cartId, Guid propertyId, int qty);
    Task RemoveItemAsync(Guid cartId, Guid propertyId);
}

// Infrastructure / Repositories / CartRepository.cs
public class CartRepository : ICartRepository
{
    private readonly RealEstateDbContext _db;

    public CartRepository(RealEstateDbContext db)
    {
        _db = db;
    }

    public async Task<Cart> CreateCartAsync()
    {
        var cart = new Cart { Id = Guid.NewGuid(), CreatedOn = DateTime.UtcNow };
        _db.Carts.Add(cart);
        await _db.SaveChangesAsync();
        return cart;
    }

    public async Task<Cart> GetCartAsync(Guid cartId)
    {
        return await _db.Carts
            .Include(c => c.Items)
              .ThenInclude(i => i.Property)
            .FirstOrDefaultAsync(c => c.Id == cartId);
    }

    public async Task AddOrUpdateItemAsync(Guid cartId, Guid propertyId, int qty)
    {
        var cart = await GetCartAsync(cartId);
        if (cart == null)
        {
            cart = await CreateCartAsync();
        }

        var existing = cart.Items.FirstOrDefault(i => i.PropertyId == propertyId);
        if (existing != null)
        {
            existing.Quantity += qty;
        }
        else
        {
            cart.Items.Add(new CartItem
            {
                Id = Guid.NewGuid(),
                CartId = cart.Id,
                PropertyId = propertyId,
                Quantity = qty
            });
        }
        await _db.SaveChangesAsync();
    }

    public async Task RemoveItemAsync(Guid cartId, Guid propertyId)
    {
        var cart = await GetCartAsync(cartId);
        if (cart == null) return;

        var existing = cart.Items.FirstOrDefault(i => i.PropertyId == propertyId);
        if (existing != null)
        {
            _db.Remove(existing);
            await _db.SaveChangesAsync();
        }
    }
}

public class CartService : ICartService
{
    private readonly ICartRepository _repo;

    public CartService(ICartRepository repo)
    {
        _repo = repo;
    }

    public async Task<Guid> AddToCartAsync(AddToCartRequest request)
    {
        await _repo.AddOrUpdateItemAsync(request.CartId, request.PropertyId, request.Quantity);
        return request.CartId;
    }

    public async Task RemoveFromCartAsync(Guid cartId, Guid propertyId)
    {
        await _repo.RemoveItemAsync(cartId, propertyId);
    }

    public async Task<CartDto> GetCartAsync(Guid cartId)
    {
        var cart = await _repo.GetCartAsync(cartId);
        if (cart == null)
        {
            return new CartDto
            {
                CartId = cartId,
                Items = new List<CartItemDto>()
            };
        }

        var dto = new CartDto
        {
            CartId = cart.Id,
            Items = cart.Items.Select(i => new CartItemDto
            {
                PropertyId = i.PropertyId,
                Title = i.Property.Title,
                Price = i.Property.Price,
                Quantity = i.Quantity
            }).ToList()
        };
        return dto;
    }
}

import React, { createContext, useState, useEffect } from 'react';
import AsyncStorage from '@react-native-async-storage/async-storage';
import { api } from './api';

export const CartContext = createContext();

export const CartProvider = ({ children }) => {
  const [cartId, setCartId] = useState(null);
  const [cart, setCart] = useState({ items: [] });

  useEffect(() => {
    (async () => {
      const stored = await AsyncStorage.getItem('cartId');
      if (stored) {
        setCartId(stored);
        await loadCart(stored);
      } else {
        // optionally create new cart on first use: but our backend “add to cart” will accept null
      }
    })();
  }, []);

  async function loadCart(id) {
    try {
      const resp = await api.get(`/cart/${id}`);
      setCart(resp.data);
    } catch (err) {
      console.error('Load cart error', err);
    }
  }

  async function addToCart(propertyId, qty = 1) {
    const req = {
      cartId: cartId || '00000000-0000-0000-0000-000000000000',  // dummy if null, backend should handle
      propertyId,
      quantity: qty
    };
    const resp = await api.post('/cart/add', req);
    const newCartId = resp.data.cartId;
    if (!cartId) {
      setCartId(newCartId);
      await AsyncStorage.setItem('cartId', newCartId);
    }
    await loadCart(newCartId);
  }

  async function removeFromCart(propertyId) {
    if (!cartId) return;
    await api.delete(`/cart/${cartId}/remove/${propertyId}`);
    await loadCart(cartId);
  }

  return (
    <CartContext.Provider value={{ cart, addToCart, removeFromCart }}>
      {children}
    </CartContext.Provider>
  );
};

  const renderItem = ({ item }) => (
    <View style={styles.itemRow}>
      <View>
        <Text style={styles.itemTitle}>{item.title}</Text>
        <Text>Price: {item.price}</Text>
        <Text>Qty: {item.quantity}</Text>
      </View>
      <TouchableOpacity
        style={styles.removeBtn}
        onPress={() => removeFromCart(item.propertyId)}
      >
        <Text style={{ color: 'white' }}>Remove</Text>
      </TouchableOpacity>
    </View>
  );

  return (
    <View style={styles.container}>
      {cart.items.length === 0 ? (
        <Text>Your cart is empty.</Text>
      ) : (
        <>
          <FlatList
            data={cart.items}
            keyExtractor={i => i.propertyId.toString()}
            renderItem={renderItem}
          />
          <View style={styles.totalRow}>
            <Text style={styles.totalText}>
              Total Items: {cart.items.reduce((a,b) => a + b.quantity, 0)}
            </Text>
          </View>
        </>
      )}
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, padding: 16 },

    marginVertical: 8,
    borderRadius: 8,
    elevation: 2
  },
  itemTitle: { fontSize: 18, fontWeight: 'bold' },
  removeBtn: {

    padding: 12,
    borderRadius: 8,
    justifyContent: 'center'
  },
  totalRow: {

});

🧪 How It Works End-to-End

On first use, the React Native app checks if a cartId is stored in AsyncStorage. If yes, it loads the cart from backend.

When user taps Add to Cart on a property detail screen:

The app calls POST /api/cart/add with { cartId, propertyId, quantity }.

Backend adds or updates cart item, returns cartId.

App updates stored cartId (if new) and reloads the cart.

The app shows cart items via GET /api/cart/{cartId}.

Remove is similarly done via DELETE /api/cart/{cartId}/remove/{propertyId}.
